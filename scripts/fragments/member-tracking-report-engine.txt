function normalizeReportTier(value){const raw=String(value||'').toLowerCase();if(raw==='research_pro'||raw==='research_pro_9')return'research_pro_9';if(raw==='intelligence'||raw==='intelligence_6')return'intelligence_6';if(raw==='supporter'||raw==='supporter_3')return'supporter_3';return'registered'}
function reportTierRank(value){return{registered:1,supporter_3:2,intelligence_6:3,research_pro_9:4}[normalizeReportTier(value)]||1}
function reportBlueprint(member){return reportTierBlueprints[normalizeReportTier(member&&member.effective_tier||member&&member.tier)]||reportTierBlueprints.registered}
function defaultReportProfile(member){const tier=normalizeReportTier(member&&member.effective_tier||member&&member.tier);const blueprint=reportTierBlueprints[tier];return{member_id:member&&member.id||'',delivery_mode:blueprint.deliveryMode,focus_mode:'tracked_first',depth:'automatic',max_items:blueprint.maxItems,include_speculation:blueprint.speculation?1:0,include_counter_evidence:blueprint.counterEvidence?1:0,include_source_ledger:blueprint.sourceLedger?1:0,include_missing_records:tier==='registered'?0:1,include_change_triggers:reportTierRank(tier)>=3?1:0,tier,blueprint}}
async function memberReportProfile(env,member){const fallback=defaultReportProfile(member);const row=await first(env.MEMBERS_DB.prepare('SELECT * FROM member_report_profiles WHERE member_id=? LIMIT 1').bind(member.id));if(!row)return fallback;return{...fallback,...row,max_items:Math.max(3,Math.min(fallback.blueprint.maxItems,Number(row.max_items||fallback.max_items))),include_speculation:reportTierRank(fallback.tier)>=3&&Boolean(row.include_speculation)?1:0,include_counter_evidence:reportTierRank(fallback.tier)>=2&&Boolean(row.include_counter_evidence)?1:0,include_source_ledger:reportTierRank(fallback.tier)>=4&&Boolean(row.include_source_ledger)?1:0,include_missing_records:reportTierRank(fallback.tier)>=2&&Boolean(row.include_missing_records)?1:0,include_change_triggers:reportTierRank(fallback.tier)>=3&&Boolean(row.include_change_triggers)?1:0}}

async function memberTrackingContext(env,memberId){const [follows,watches]=await Promise.all([all(env.MEMBERS_DB.prepare('SELECT entity_id AS target_id,entity_type AS target_type,label,route,notifications_enabled AS active,minimum_tier,updated_at FROM member_entity_follows WHERE member_id=? AND notifications_enabled=1 ORDER BY updated_at DESC LIMIT 200').bind(memberId)),all(env.MEMBERS_DB.prepare('SELECT target_id,target_type,label,route,active,minimum_tier,criteria_json,updated_at FROM member_watch_items WHERE member_id=? AND active=1 ORDER BY updated_at DESC LIMIT 200').bind(memberId))]);const items=[...watches.map(item=>({...item,source:'watchlist'})),...follows.map(item=>({...item,source:'follow'}))];const labels=[...new Set(items.map(item=>clean(item.label,180)).filter(Boolean))];return{items,labels,count:items.length,followCount:follows.length,watchCount:watches.length}}
function trackingTerms(context){const values=[];for(const item of context.items||[]){values.push(item.label,item.target_id,item.target_type);try{const criteria=JSON.parse(item.criteria_json||'{}');for(const value of Object.values(criteria||{})){if(Array.isArray(value))values.push(...value);else if(typeof value==='string'||typeof value==='number')values.push(value)}}catch{}}return[...new Set(values.flatMap(value=>clean(value,240).toLowerCase().split(/[^a-z0-9]+/)).filter(token=>token.length>=3))]}
function reportCandidateRows(bundle){const rows=[];if(typeof developmentRows==='function')for(const item of developmentRows(bundle))rows.push({type:'development',title:item.heading,summary:item.summary,status:item.status,lane:item.lane,route:item.route});if(typeof actorRows==='function')for(const item of actorRows(bundle))rows.push({type:'actor',title:item.name,summary:`${item.role}${item.context?` — ${item.context}`:''}`,status:'documented role',lane:'named actor or institution',route:item.route});if(typeof sectionRows==='function')for(const item of sectionRows(bundle))rows.push({type:'analysis',title:item.headline||item.heading,summary:`${item.meaning}${item.likely?` Likely next: ${item.likely}`:''}`,status:'analytic inference',lane:item.heading,route:''});if(typeof scenarioRows==='function')for(const item of scenarioRows(bundle))rows.push({type:'scenario',title:item.heading,summary:item.trajectory,status:item.band,lane:'speculation',route:'',needed:item.needed||[],weakens:item.weakens||[],boundary:item.boundary||''});if(typeof recordRows==='function')for(const item of recordRows(bundle))rows.push({type:'record',title:item,summary:'Priority record or missing evidence required to advance the investigation.',status:'record required',lane:'records',route:'record-intake-queue.html'});return rows}
function scoreTrackedRow(row,terms,labels){const hay=clean(`${row.title||''} ${row.summary||''} ${row.status||''} ${row.lane||''} ${row.route||''}`,3000).toLowerCase();let score=0;for(const label of labels){const normalized=clean(label,180).toLowerCase();if(normalized&&hay.includes(normalized))score+=18}for(const term of terms)if(hay.includes(term))score+=2;return score}
function segmentAllowedByTier(member,segmentKey,profile){if(profile.delivery_mode==='all_selected')return true;const tier=normalizeReportTier(member.effective_tier||member.tier);if(segmentKey==='registered_release_notices')return true;if(tier==='research_pro_9')return segmentKey==='research_pro_reports';if(tier==='intelligence_6')return['intelligence_daily_member_brief','supporter_weekly_member_brief'].includes(segmentKey);if(tier==='supporter_3')return['public_daily_brief','supporter_weekly_member_brief'].includes(segmentKey);return['public_daily_brief','public_weekly_digest'].includes(segmentKey)}
function reportTierTitle(tier,kind){const key=normalizeReportTier(tier);if(key==='research_pro_9')return kind==='weekly'?'Research Pro Evidence Dossier':'Research Pro Change Report';if(key==='intelligence_6')return kind==='weekly'?'Intelligence Weekly Synthesis':'Intelligence Watch Brief';if(key==='supporter_3')return kind==='weekly'?'Supporter Intelligence Review':'Daily Control Brief';return kind==='weekly'?'Weekly Signal Drop':'Daily Control Brief'}
async function recordPersonalReport(env,member,campaign,segmentKey,profile,tracking,included,subject,contentHash){await env.MEMBERS_DB.prepare(`INSERT OR IGNORE INTO member_report_runs (id,member_id,campaign_id,report_kind,report_date,effective_tier,segment_key,tracking_count,included_item_count,subject,content_hash,status,created_at) VALUES (?,?,?,?,?,?,?,?,?,?,?,'queued',?)`).bind(id(env,'member-report'),member.id,campaign.id,campaign.kind,iso(env).slice(0,10),normalizeReportTier(member.effective_tier||member.tier),segmentKey,tracking.count,included,clean(subject,180),contentHash,iso(env)).run().catch(()=>null)}

The system did not invent a match; the general report remains below.</p>';const profileSummary=`${blueprint.label} · ${profile.depth} depth · ${profile.focus_mode.replace(/_/g,' ')} · maximum ${profile.max_items} personalised items`;const trackedBlock=`<section style="border:1px solid #8d7137;border-radius:14px;padding:18px;margin:18px 0;background:#090805"><p style="margin:0;color:#d8b56a;font-size:12px;letter-spacing:.08em;text-transform:uppercase">Your tracked intelligence · ${html(TRACKED_REPORT_ENGINE_VERSION)}</p><h2 style="margin:7px 0">${html(title)}</h2><p style="color:#b9aa82">${html(profileSummary)}</p><p><strong>Tracking:</strong> ${trackedNames.length?html(trackedNames.join(' · ')):'No personal tracking targets yet — add followed topics or a watchlist to personalise this section.'}</p>${rowsHtml}${profile.include_missing_records?'<p style="color:#b9aa82"><strong>Research rule:</strong> missing records stay visible until obtained, contradicted or deliberately removed from the watch.</p>':''}${profile.include_source_ledger?'<p><a href="https://matrixreprogrammed.com/evidence-vault.html" style="color:#f0ca78">Open source ledger and evidence routes</a></p>':''}</section>`;let htmlContent=String(base.htmlContent||'');const executiveMarker='<div style="border-left:4px solid #d8b56a';htmlContent=htmlContent.includes(executiveMarker)?htmlContent.replace(executiveMarker,`${trackedBlock}${executiveMarker}`):htmlContent.replace(/(<h1[^>]*>[\s\S]*?<\/h1>)/i,`$1${trackedBlock}`);const textRows=selected.length?selected.map(item=>`- [${item.type}] ${item.title}: ${item.summary}${profile.include_counter_evidence&&item.status?`\n  Status/boundary: ${item.status}`:''}${item.type==='scenario'?`\n  Strengthen: ${(item.needed||[]).join(';
