function arrayValue(value){return Array.isArray(value)?value.filter(Boolean):[]}
function objectValue(value){return value&&typeof value==='object'&&!Array.isArray(value)?value:{}}
function firstArray(object,keys){for(const key of keys){const value=objectValue(object)[key];if(Array.isArray(value)&&value.length)return value}return[]}
function uniqueRows(rows,key='heading'){const seen=new Set();return rows.filter(row=>{const value=clean(row&&row[key]||'',240).toLowerCase();if(!value||seen.has(value))return false;seen.add(value);return true})}

function developmentRows(bundle){const daily=objectValue(bundle.daily);const drops=objectValue(bundle.drops);const live=objectValue(bundle.live);const candidates=[...arrayValue(daily.namedDevelopments),...arrayValue(daily.topSignals),...firstArray(drops,['drops','items','records','entries','latest']),...firstArray(live,['items','records','findings','entries'])];return uniqueRows(candidates.map((item,index)=>{item=typeof item==='string'?{title:item}:objectValue(item);return{id:clean(item.id||item.signalId||item.canonicalId||`development-${index+1}`,160),heading:clean(item.title||item.headline||item.subject||item.label||`Development ${index+1}`,220),summary:clean(item.why||item.whyItMatters||item.summary||item.description||item.finding||item.body||'',760),status:clean(item.status||item.sourceStatus||item.evidenceLevel||item.confidence||'',180),lane:clean(item.lane||item.category||item.topic||'',120),route:clean(item.route||item.url||item.publicRoute||'',300)}}),'heading').slice(0,6)}

function scenarioRows(bundle){const command=objectValue(bundle.command);const synthesis=objectValue(bundle.synthesis);const daily=objectValue(bundle.daily);const speculative=objectValue(synthesis.speculativeLayer);const candidates=[...arrayValue(command.scenarioMatrix),...arrayValue(daily.speculationDesk),...arrayValue(speculative.scenarios)];return uniqueRows(candidates.map((scenario,index)=>{scenario=objectValue(scenario);return{id:clean(scenario.id||scenario.scenarioId||`scenario-${index+1}`,120),heading:clean(scenario.title||scenario.name||`Scenario ${index+1}`,220),band:clean(scenario.plausibilityBand||scenario.band||scenario.status||'unrated',120),trajectory:clean(scenario.trajectory||scenario.summary||scenario.meaning||scenario.description||'',800),needed:arrayValue(scenario.evidenceNeeded||scenario.recordsNeeded||scenario.confirmationTriggers).map(item=>clean(item,240)).filter(Boolean).slice(0,5),weakens:arrayValue(scenario.disconfirmingEvidence||scenario.counterSignals||scenario.downgradeConditions).map(item=>clean(item,240)).filter(Boolean).slice(0,5),boundary:clean(scenario.boundary||scenario.evidenceBoundary||'',520)}}),'heading').slice(0,4)}
function recordRows(bundle){const daily=objectValue(bundle.daily);const command=objectValue(bundle.command);const rows=[...arrayValue(daily.missingRecords).map(item=>typeof item==='string'?item:`${clean(item.section||'Record',100)}: ${clean(item.record||item.title||item.name,320)}`),...arrayValue(command.watchNext)];return [...new Set(rows.map(item=>clean(item,420)).filter(Boolean))].slice(0,8)}
function watchRows(bundle){const daily=objectValue(bundle.daily);const synthesis=objectValue(bundle.synthesis);return [...new Set([...arrayValue(daily.tomorrowWatchList),...arrayValue(objectValue(synthesis.uncertaintyLayer).watchNext)].map(item=>clean(item,420)).filter(Boolean))].slice(0,7)}
function routeUrl(request,route,fallback){const base=origin(request);const value=clean(route||'',400);if(/^https?:\/\//i.test(value))return value;if(value)return new URL(value.replace(/^\/+/,''),base+'/').toString();return new URL(fallback,base+'/').toString()}
